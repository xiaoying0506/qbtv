let isConverting = false; convertBtn.addEventListener('click', async () => { if (!selectedFile || isConverting) return; isConverting = true; convertBtn.disabled = true; convertBtn.classList.add('opacity-50', 'cursor-not-allowed'); progressArea.classList.remove('hidden'); resultArea.classList.add('hidden'); const targetFormat = document.getElementById('targetFormat').value; const targetBitrate = document.getElementById('targetBitrate').value; try { if (!ffmpeg.isLoaded()) { statusText.innerText = '正在加载转换引擎...'; await ffmpeg.load(); } ffmpeg.setProgress(({ ratio }) => { const percent = Math.min(Math.round(ratio * 100), 100); progressBar.style.width = percent + '%'; progressPercent.innerText = percent + '%'; statusText.innerText = percent < 100 ? '正在转码音频格式中...' : '正在打包生成目标文件...'; }); const inputExt = selectedFile.name.split('.').pop(); const inputName = `input_${Date.now()}.${inputExt}`; const outputName = `output_${Date.now()}.${targetFormat}`; statusText.innerText = '正在读取文件...'; ffmpeg.FS('writeFile', inputName, await fetchFile(selectedFile)); statusText.innerText = '正在转码...'; let ffmpegArgs = ['-i', inputName]; if (['mp3', 'aac', 'ogg', 'm4a'].includes(targetFormat)) { ffmpegArgs.push('-b:a', targetBitrate); } ffmpegArgs.push(outputName); await ffmpeg.run(...ffmpegArgs); const data = ffmpeg.FS('readFile', outputName); const blob = new Blob([data.buffer], { type: `audio/${targetFormat}` }); const url = URL.createObjectURL(blob); const originalBaseName = selectedFile.name.substring(0, selectedFile.name.lastIndexOf('.')); const finalFileName = `${originalBaseName}_converted.${targetFormat}`; audioPreview.src = url; downloadBtn.href = url; downloadBtn.download = finalFileName; progressArea.classList.add('hidden'); resultArea.classList.remove('hidden'); try { ffmpeg.FS('unlink', inputName); ffmpeg.FS('unlink', outputName); } catch(e) {} } catch (err) { alert('转换失败: ' + err.message); console.error(err); } finally { isConverting = false; convertBtn.disabled = false; convertBtn.classList.remove('opacity-50', 'cursor-not-allowed'); } });
